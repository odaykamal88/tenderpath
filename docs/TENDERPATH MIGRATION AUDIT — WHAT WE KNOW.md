function computeGrantReadinessScore(orgType, aiAnalysis, devThemes) {  
  const score \= aiAnalysis.overall\_score || 50;  
  const certs \= (aiAnalysis.certifications\_detected || \[\]).map(c \=\> c.toLowerCase());  
  const services \= (aiAnalysis.services || \[\]).map(s \=\> s.toLowerCase()).join(' ');  
  const capabilities \= (aiAnalysis.capabilities || \[\]).map(c \=\> c.toLowerCase()).join(' ');  
  const allText \= \`${services} ${capabilities}\`;

  // Impact Methodology (30%) — signals that org understands impact measurement  
  let impactScore \= 30;  
  if (allText.includes('impact') || allText.includes('outcomes') || allText.includes('theory of change') || allText.includes('results')) impactScore \= 80;  
  else if (allText.includes('evaluation') || allText.includes('assessment') || allText.includes('monitoring')) impactScore \= 60;  
  else if (orgType \=== 'research\_institution') impactScore \= 70;  
  else if (orgType \=== 'NGO') impactScore \= 55;  
  else if (orgType \=== 'SME') impactScore \= 40;  
  const impactWeighted \= Math.round(impactScore \* 0.30);

  // M\&E Frameworks (20%) — monitoring & evaluation capability  
  let meScore \= 25;  
  if (allText.includes('monitoring and evaluation') || allText.includes('m\&e') || allText.includes('monitoring & evaluation')) meScore \= 85;  
  else if (allText.includes('data collection') || allText.includes('reporting') || allText.includes('kpi') || allText.includes('metrics')) meScore \= 65;  
  else if (orgType \=== 'research\_institution') meScore \= 70;  
  else if (orgType \=== 'NGO') meScore \= 50;  
  const meWeighted \= Math.round(meScore \* 0.20);

  // Past Grant Management Experience (20%) — track record with grants  
  let grantMgmtScore \= 30;  
  if (allText.includes('grant management') || allText.includes('donor funded') || allText.includes('grant funded') || allText.includes('iNGO')) grantMgmtScore \= 85;  
  else if (allText.includes('project management') || allText.includes('programme management') || allText.includes('development')) grantMgmtScore \= 60;  
  else if (orgType \=== 'NGO') grantMgmtScore \= 55;  
  else if (orgType \=== 'research\_institution') grantMgmtScore \= 50;  
  else if (orgType \=== 'cooperative') grantMgmtScore \= 45;  
  else if (orgType \=== 'SME') grantMgmtScore \= 35;  
  const grantMgmtWeighted \= Math.round(grantMgmtScore \* 0.20);

  // Organizational Capacity (15%) — scale from tender readiness dimensions  
  const orgCapDim \= aiAnalysis.readiness\_dimensions?.technical\_capability || score;  
  const orgCapScore \= orgCapDim;  
  const orgCapWeighted \= Math.round(orgCapScore \* 0.15);

  // Financial Controls (15%) — financial management signals  
  let finScore \= 35;  
  if (certs.some(c \=\> c.includes('iso') || c.includes('audit') || c.includes('account'))) finScore \= 85;  
  else if (aiAnalysis.readiness\_dimensions?.financial\_stability \>= 60\) finScore \= 70;  
  else if (aiAnalysis.readiness\_dimensions?.financial\_stability \>= 40\) finScore \= 55;  
  else finScore \= Math.min(65, aiAnalysis.readiness\_dimensions?.financial\_stability || 35);  
  const finWeighted \= Math.round(finScore \* 0.15);

  const total \= Math.min(100, impactWeighted \+ meWeighted \+ grantMgmtWeighted \+ orgCapWeighted \+ finWeighted);

  return {  
    grantReadinessScore: total,  
    grantDimensions: {  
      impact\_methodology: impactScore,  
      monitoring\_evaluation: meScore,  
      grant\_management: grantMgmtScore,  
      organizational\_capacity: orgCapScore,  
      financial\_controls: finScore  
    }  
  };  
}

### **SME Fit Score (computeSmeFitScore)**

function computeSmeFitScore(tender, aiAnalysis) {  
  let score \= 50; // base  
  const companySize \= aiAnalysis?.company\_size\_signal || 'small';

  // Contract value scoring  
  const val \= parseInt((tender.estimated\_value || '').replace(/\[^0-9\]/g, '')) || 0;  
  if (val \> 0 && val \<= 200000\) score \+= 20;  
  else if (val \> 200000 && val \<= 500000\) score \+= 10;  
  else if (val \> 1000000\) score \-= 10;

  // Registration complexity (1-5, lower is better)  
  const regComp \= tender.registration\_complexity || 2;  
  if (regComp \<= 2\) score \+= 10;  
  else if (regComp \>= 4\) score \-= 15;

  // Experience level  
  const expLevel \= tender.experience\_level || 'any';  
  if (expLevel \=== 'any' || expLevel \=== 'entry') score \+= 10;  
  else if (expLevel \=== 'senior') score \-= 10;

  if (tender.jv\_allowed) score \+= 5;  
  if ((tender.submission\_language || 'English').toLowerCase() \=== 'english') score \+= 5;  
  if (companySize \=== 'micro' && val \> 500000\) score \-= 15;  
  if (companySize \=== 'large' && val \< 100000\) score \-= 5;

  score \= Math.max(0, Math.min(100, score));  
  // Labels: \>= 65 → 'High Fit', \>= 40 → 'Medium Fit', else → 'Requires Registration'  
}
